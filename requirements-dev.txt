# Pinned test toolchain — keeps the CI matrix (Python 3.10 / 3.11 / 3.12) hermetic.
#
# Exact pins so pip resolves an IDENTICAL set on every interpreter. This closes the
# class of failure where an unpinned transitive dependency (e.g. pytest-base-url,
# which ships a session-scoped `base_url` fixture) resolves onto one runner and not
# the others, diverging the matrix. Runtime deps stay loose in requirements.txt;
# this file is dev/CI only.
#
# All four have universal wheels and support 3.10–3.12.
# Versions chosen to clear the six advisories open on the default branch
# (2026-07-27). Each is at or above the first fixed release:
#   requests  CVE-2024-47081 (.netrc credential leak)        fixed 2.32.4
#   requests  CVE-2026-25645 (temp file reuse, zipped paths) fixed 2.33.0
#   pytest    CVE-2025-71176 (tmpdir handling)               fixed 9.0.3
#   PyNaCl    CVE-2025-69277 (libsodium disallowed inputs)   fixed 1.6.2
#   flask     CVE-2025-47278 (fallback signing key)          fixed 3.1.1
#   flask     CVE-2026-27205 (session missing Vary: Cookie)  fixed 3.1.3
pytest==9.1.1
requests==2.34.2
flask==3.1.3
PyNaCl==1.6.2
